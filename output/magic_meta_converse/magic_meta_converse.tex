\begin{abstract}
Magic-state distillation and resource-state synthesis are central primitives for fault-tolerant universal quantum computation. We prove a general \emph{one-shot} meta-converse that upper bounds any achievable yield of a target magic state under stabilizer-preserving operations (SPO), expressed via the hypothesis-testing divergence to the stabilizer polytope. Concretely, for any protocol that outputs a $k$-fold target state $\omega^{\otimes k}$ within error $\varepsilon$, we show
\[
R_h^{2\varepsilon}\big(\omega^{\otimes k}\big)\ \le\ R_h^{\varepsilon}(\rho),
\]
where $R_h^{\varepsilon}(\cdot):=\inf_{\sigma\in\mathrm{STAB}} \Dh{\varepsilon}(\,\cdot\,\Vert\sigma)$ and $\mathrm{STAB}$ denotes the convex set of stabilizer states. The bound is dimension-free, applies to arbitrary finite-dimensional systems, and immediately yields (i) finite-blocklength converse bounds on distillation yield and synthesis cost, and (ii) an asymptotic \emph{strong-converse} upper bound on the distillable magic rate in terms of the (regularized) relative entropy of magic. In small instances, the inner optimization admits LP/SDP formulations; in larger instances, cutting-plane methods with a stabilizer-oracle provide practical evaluations.
\end{abstract}

\section{Introduction}
Magic states quantify the non-stabilizerness that enables universal quantum computation beyond Clifford operations. In practice, one prepares noisy resource states and distills high-fidelity magic via fault-tolerant stabilizer protocols, or synthesizes arbitrary states from magic under stabilizer operations. Precise \emph{finite-blocklength} converse bounds remain highly valuable: they certify the limits of NISQ-era and fault-tolerant pipelines, inform architecture choices, and guide algorithmic tradeoffs.
\medskip

We provide a concise, general meta-converse for magic-state distillation and synthesis in the \emph{one-shot} regime, based on quantum hypothesis testing against the stabilizer polytope. The proof leverages two elementary ingredients: (i) data processing for hypothesis-testing divergence and (ii) a smoothing monotonicity under trace-distance perturbations. From these, we obtain dimension-free, model-agnostic converse bounds and a strong-converse asymptotic corollary.

\subsection*{Notation}
For a finite-dimensional Hilbert space $\calH$, $\calD(\calH)$ is the set of density operators on $\calH$. The set $\mathrm{STAB}$ consists of convex combinations of pure stabilizer states on $\calH$; we write $\mathrm{STAB}\subset\calD(\calH)$ and call its elements \emph{free states}. A completely positive trace-preserving (CPTP) map $\Lambda$ is \emph{stabilizer-preserving} (SPO) if $\Lambda(\mathrm{STAB})\subseteq\mathrm{STAB}$. The fidelity is $\F(\rho,\sigma):=\|\sqrt{\rho}\sqrt{\sigma}\|_1^2$, and $\tfrac12\|\rho-\sigma\|_1\le\sqrt{1-\F(\rho,\sigma)}$.

\section{Preliminaries}
\begin{definition}[Hypothesis-testing divergence]
For $\rho,\sigma\in\calD(\calH)$ and $\varepsilon\in[0,1)$, set
\begin{equation}\label{eq:Dh}
\Dh{\varepsilon}(\rho\Vert\sigma)\ :=\ -\log \beta_\varepsilon(\rho\Vert\sigma),\qquad
\beta_\varepsilon(\rho\Vert\sigma)\ :=\ \min_{\substack{0\le \Lambda \le \id\\ \Tr[\Lambda\rho]\ge 1-\varepsilon}}\ \Tr[\Lambda\sigma].
\end{equation}
\end{definition}
Two standard properties will be used repeatedly: (i) \emph{data processing}: $\Dh{\varepsilon}(\Phi(\rho)\Vert\Phi(\sigma))\le \Dh{\varepsilon}(\rho\Vert\sigma)$ for any channel $\Phi$; and (ii) \emph{monotonicity in $\varepsilon$}: $\Dh{\varepsilon}(\rho\Vert\sigma)$ is nonincreasing in $\varepsilon$.

\begin{definition}[Hypothesis-testing magic]\label{def:Rhe}
For $\varepsilon\in[0,1)$ and $\rho\in\calD(\calH)$, define
\begin{equation}
R_h^{\varepsilon}(\rho)\ :=\ \inf_{\sigma\in \mathrm{STAB}} \Dh{\varepsilon}(\rho\Vert\sigma).
\end{equation}
\end{definition}
This quantity is a one-shot converse monotone: it upper bounds what can be produced from $\rho$ via SPO in the sense of Theorem~\ref{thm:meta}. We also recall the (Umegaki) relative entropy of magic,
\begin{equation}\label{eq:REM}
R(\rho)\ :=\ \inf_{\sigma\in\mathrm{STAB}} D(\rho\Vert\sigma),\qquad
D(\rho\Vert\sigma):=\Tr[\rho(\log\rho-\log\sigma)],
\end{equation}
which governs the asymptotics.

\begin{lemma}[Monotonicity under SPO]\label{lem:monotone}
For any SPO $\Lambda$ and any $\varepsilon\in[0,1)$, one has $R_h^{\varepsilon}(\Lambda(\rho))\ \le\ R_h^{\varepsilon}(\rho)$.
\end{lemma}
\begin{proof}
For any $\tau\in\mathrm{STAB}$, $\Lambda(\tau)\in\mathrm{STAB}$ and by data processing
$\Dh{\varepsilon}(\Lambda(\rho)\Vert \Lambda(\tau))\le \Dh{\varepsilon}(\rho\Vert\tau)$.
Taking the infimum over $\tau\in\mathrm{STAB}$ proves the claim.
\end{proof}

\begin{lemma}[Smoothing monotonicity]\label{lem:smooth}
Let $\rho,\rho'\in\calD(\calH)$ with $\tfrac12\|\rho-\rho'\|_1\le\delta$.
Then for all $\varepsilon\in[0,1-\delta)$ and all $\sigma\in\calD(\calH)$,
\begin{equation}\label{eq:smooth}
\Dh{\varepsilon}(\rho\Vert\sigma)\ \ge\ \Dh{\varepsilon+\delta}(\rho'\Vert\sigma).
\end{equation}
Consequently, $R_h^{\varepsilon}(\rho)\ \ge\ R_h^{\varepsilon+\delta}(\rho')$.
\end{lemma}
\begin{proof}
If $\Tr[\Lambda\rho']\ge 1-(\varepsilon+\delta)$ then $\Tr[\Lambda\rho]\ge \Tr[\Lambda\rho']-\|\rho-\rho'\|_1/2 \ge 1-\varepsilon$ by Hölder, hence the feasible set for $\beta_{\varepsilon+\delta}(\rho'\Vert\sigma)$ is included in that of $\beta_\varepsilon(\rho\Vert\sigma)$. The inequality for $R_h$ follows by taking infima over $\sigma\in\mathrm{STAB}$.
\end{proof}

\section{One-shot meta-converse for distillation and synthesis}
We formulate distillation with target state $\omega$ and synthesis as its reverse transformation.

\begin{definition}[One-shot distillable yield and synthesis cost]
Fix a target magic state $\omega$ on $\calH_\mathrm{out}$. For input $\rho$ on $\calH_\mathrm{in}$ and error $\varepsilon\in[0,1)$, define the achievable \emph{distillation yield} $k_\varepsilon^\star(\rho\!\to\!\omega)$ as the largest $k\in\bbN$ for which there exists an SPO $\Lambda$ such that
\begin{equation}
\tfrac12\big\|\Lambda(\rho)-\omega^{\otimes k}\big\|_1\ \le\ \varepsilon.
\end{equation}
Dually, the \emph{synthesis cost} $m_\varepsilon^\star(\omega\!\to\!\rho)$ is the smallest $m\in\bbN$ for which some SPO $\Lambda$ satisfies $\tfrac12\|\Lambda(\omega^{\otimes m})-\rho\|_1\le\varepsilon$.
\end{definition}

\begin{theorem}[One-shot meta-converse]\label{thm:meta}
If $\tfrac12\|\Lambda(\rho)-\omega^{\otimes k}\|_1\le \varepsilon$ for some SPO $\Lambda$, then
\begin{equation}\label{eq:meta}
R_h^{2\varepsilon}\!\big(\omega^{\otimes k}\big)\ \le\ R_h^{\varepsilon}(\rho).
\end{equation}
Equivalently,
\begin{equation}\label{eq:yield-implicit}
k_\varepsilon^\star(\rho\!\to\!\omega)\ \le\ \max\Big\{k\in\bbN:\ R_h^{2\varepsilon}\!\big(\omega^{\otimes k}\big)\ \le\ R_h^{\varepsilon}(\rho)\Big\}.
\end{equation}
\end{theorem}
\begin{proof}
By Lemma~\ref{lem:smooth} with $\rho'=\omega^{\otimes k}$,
$R_h^{2\varepsilon}(\omega^{\otimes k}) \le R_h^{\varepsilon}(\Lambda(\rho))$.
Then apply Lemma~\ref{lem:monotone} to get $R_h^{\varepsilon}(\Lambda(\rho)) \le R_h^{\varepsilon}(\rho)$.
\end{proof}

\begin{corollary}[Synthesis cost converse]\label{cor:synthesis}
If $\tfrac12\|\Lambda(\omega^{\otimes m})-\rho\|_1\le \varepsilon$ for some SPO $\Lambda$, then
\begin{equation}\label{eq:synth}
R_h^{2\varepsilon}(\rho)\ \le\ R_h^{\varepsilon}\!\big(\omega^{\otimes m}\big).
\end{equation}
Hence
\begin{equation}\label{eq:cost-implicit}
m_\varepsilon^\star(\omega\!\to\!\rho)\ \ge\ \min\Big\{m\in\bbN:\ R_h^{\varepsilon}\!\big(\omega^{\otimes m}\big)\ \ge\ R_h^{2\varepsilon}(\rho)\Big\}.
\end{equation}
\end{corollary}

\section{Asymptotics and strong converse}
The following asymptotic equipartition property (AEP) is a quantum Stein-type statement against the convex composite alternative $\mathrm{STAB}$.
\begin{proposition}[AEP for magic]\label{prop:AEP}
For any fixed state $\tau$ and any $\varepsilon\in(0,1)$,
\begin{equation}\label{eq:AEP}
\lim_{n\to\infty}\ \frac1n\, R_h^{\varepsilon}\!\big(\tau^{\otimes n}\big)\ =\ R(\tau),
\end{equation}
where $R(\tau)$ is the relative entropy of magic \eqref{eq:REM}.
\end{proposition}
\begin{proof}[Proof sketch]
Upper bound: quantum Stein's lemma with composite alternatives yields $\limsup \tfrac1n \Dh{\varepsilon}(\tau^{\otimes n}\Vert\sigma^{\otimes n}) \le D(\tau\Vert\sigma)$ for fixed $\sigma$; taking infima over $\sigma\in\mathrm{STAB}$ gives $\limsup \tfrac1n R_h^{\varepsilon}(\tau^{\otimes n}) \le R(\tau)$. Lower bound: for any $\delta>0$, choose $\sigma^\star\in\mathrm{STAB}$ with $D(\tau\Vert\sigma^\star)\le R(\tau)+\delta$ and apply the strong form of Stein's lemma to obtain $\liminf \tfrac1n \Dh{\varepsilon}(\tau^{\otimes n}\Vert(\sigma^\star)^{\otimes n}) \ge D(\tau\Vert\sigma^\star)-\delta$. Taking $\delta\downarrow 0$ concludes.
\end{proof}

\begin{corollary}[Strong-converse distillation rate]\label{cor:sc}
Let $(n,k_n,\varepsilon)$-protocols distill $\omega^{\otimes k_n}$ from $\rho^{\otimes n}$ with $\sup_n \varepsilon<1$ under SPO. Then
\begin{equation}\label{eq:rate-bound}
\limsup_{n\to\infty}\ \frac{k_n}{n}\ \le\ \frac{\overline{R}(\rho)}{R(\omega)},
\end{equation}
where $\overline{R}(\rho):=\limsup_{n\to\infty}\tfrac1n R(\rho^{\otimes n})$ is the regularized relative entropy of magic. Moreover, if the rate exceeds this bound, the error necessarily tends to $1$.
\end{corollary}
\begin{proof}
Apply Theorem~\ref{thm:meta} with $\rho^{\otimes n}$ and $k=k_n$, divide by $n$, and pass to the limit using Proposition~\ref{prop:AEP}. The strong converse follows from the hypothesis-testing formulation and the standard Stein exponent argument.
\end{proof}

\section{Computability: LP/SDP and cutting planes}
For fixed $\rho$ and $\varepsilon$, the inner quantity $\beta_\varepsilon(\rho\Vert\sigma)$ in \eqref{eq:Dh} is an SDP over the test effect. Minimizing $\Tr[\Lambda\sigma]$ over $\sigma\in\mathrm{STAB}$ is a linear program because $\mathrm{STAB}$ is a polytope:
\begin{align}
\beta_\varepsilon(\rho\Vert \mathrm{STAB})\ :=\ \inf_{\sigma\in\mathrm{STAB}}\ \beta_\varepsilon(\rho\Vert\sigma)
\ =\ \min_{\substack{0\le \Lambda\le \id\\ \Tr[\Lambda\rho]\ge 1-\varepsilon}}\ \min_{\sigma\in\mathrm{STAB}}\ \Tr[\Lambda\sigma].
\end{align}
The inner minimum equals the minimum of $\Tr[\Lambda s]$ over extreme stabilizer states $s$, resulting in an SDP with exponentially many linear inequalities $\Tr[\Lambda s]\ge t$. This can be handled via a cutting-plane scheme driven by a \emph{stabilizer-oracle} that, given $\Lambda$, finds an $s$ minimizing $\Tr[\Lambda s]$. Small instances can be solved directly by enumerating stabilizer vertices; larger ones can leverage symmetry reduction and Clifford twirling.

\section{Examples and consequences}
\begin{example}[Qubit $T$ state]
Let $\omega = \lvert T\rangle\!\langle T\rvert$ on a single qubit. Then $R(\omega)>0$ and by \eqref{eq:rate-bound}, the distillable rate from i.i.d.\ input $\rho$ under SPO is strongly conversely upper bounded by $\overline{R}(\rho)/R(\omega)$. In finite blocklength, \eqref{eq:yield-implicit} quantifies the maximum $k$ permitted by $R_h^{\varepsilon}(\rho)$.
\end{example}

\begin{remark}[Relation to other magic monotones]
Since $\Dh{\varepsilon}\le \Dm$ for any fixed $\varepsilon$, we have $R_h^{\varepsilon}(\rho)\le R_{\max}(\rho)$, where $R_{\max}(\rho):=\inf_{\sigma\in\mathrm{STAB}}\Dm(\rho\Vert\sigma)=\log(1+\mathsf{RoM}(\rho))$ is the max-relative-entropy magic equal to the log-robustness. Hence Theorem~\ref{thm:meta} also yields the trivial but often handy bound $R_h^{2\varepsilon}(\omega^{\otimes k})\le R_{\max}(\rho)$.
\end{remark}

\section{Conclusion and outlook}
We established a clean one-shot meta-converse for magic-state distillation and synthesis under SPO, with finite-blocklength expressions and a strong-converse asymptotic consequence. Mathematically, the approach hinges only on hypothesis testing and convexity; operationally, it provides computable certificates for near-term protocols. Future directions include second-order (normal approximation) refinements, extensions to contextually-motivated free sets, and converse bounds under noise-biased fault-tolerant primitives.
