\section{qTPE decoupling, chaos-to-gap, $E_n$ recovery, and adiabatic channels}\label{app:qtpe}

\subsection{Notation and preliminaries}
We denote by $\|\cdot\|_{2\to 2}$ the superoperator norm induced by the Hilbert--Schmidt inner product, by $\|\cdot\|_1$ the trace norm, and by $\Mtwo_\nu$ the second-moment operator of an ensemble $\nu$ acting on $\mathcal{H}$:
\begin{equation}
\Mtwo_\nu(X)=\mathbb{E}_{U\sim\nu}\!\left[U^{\otimes 2}X(U^\dagger)^{\otimes 2}\right],\qquad
\Pi_{\rm Haar}=\frac{1}{d(d\!+\!1)}\!\left(\operatorname{id}+\operatorname{Ad}_{\mathsf{F}}\right)\otimes\mathbb{1}_{\operatorname{Comm}},
\end{equation}
where $d=\dim\mathcal{H}$ and $\operatorname{Ad}_{\mathsf{F}}(X)=\mathsf{F}X\mathsf{F}$. On the orthogonal complement of $\operatorname{span}\{\mathbb{1},\mathsf{F}\}$ we decompose $\Mtwo_\nu=\Pi_{\rm Haar}+K_\nu$ and write $\delta(\nu):=\|K_\nu\|_{2\to 2}$.

\paragraph{qTPE.} We say $\nu$ is a $(\gamma,2)$ quantum tensor product expander (qTPE) if $\delta(\nu)\le 1-\gamma$.

\subsection{Decoupling with approximate second moments}\label{app:qtpe-decoupling}
We prove Theorem~\ref{thm:qtpe-page} in a one-shot setting, following the swap-trick method but keeping the (small) second-moment defect explicit.

\begin{lemma}[One-step decoupling with second-moment defect]\label{lem:one-step-decouple}
Let $\rho_{MA}$ be a state on $M\!\otimes\!A$ and $V:A\!\to\!RB$ an isometry. For an ensemble $\nu$ on $A$ with defect $\delta(\nu)$,
\begin{align}
\mathbb{E}_{U\sim\nu}\left\| \operatorname{Tr}_B\!\left[ VU\ \rho_{MA}\ U^\dagger V^\dagger \right] - \rho_M\otimes\frac{\mathbb{1}_R}{d_R} \right\|_1
\ \le\ 2^{-\frac12\!\left(H_2(M|A)_\rho-\log d_R\right)}\ +\ \delta(\nu).\label{eq:one-step}
\end{align}
\end{lemma}
\begin{proof}
Let $\Delta:=\operatorname{Tr}_B[VU\,\rho_{MA}\,U^\dagger V^\dagger]-\rho_M\otimes\frac{\mathbb{1}_R}{d_R}$. By $\|\cdot\|_1\le \sqrt{d_Md_R}\,\|\cdot\|_2$ and the swap trick,
\begin{align}
\mathbb{E}\|\Delta\|_2^2
&= \operatorname{Tr}\!\left[\big(\Mtwo_\nu-\Pi_{\rm Haar}\big)(X)\,Y\right]\ +\ \operatorname{Tr}\!\left[\Pi_{\rm Haar}(X)\,Y\right],\nonumber
\end{align}
where $X$ and $Y$ are the usual $2$-copy ``Choiized'' tensors built from $\rho_{MA}$ and $V$ (explicit formulas omitted for brevity). The Haar term equals $2^{-(H_2(M|A)_\rho-\log d_R)}$. The defect term is bounded by $\delta(\nu)\cdot\|X\|_2\|Y\|_2\le \delta(\nu)$ after normalization. Taking square roots yields~\eqref{eq:one-step}.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:qtpe-page}]
Iterate Lemma~\ref{lem:one-step-decouple} for the comb, conditioning on the past and using the triangle inequality to sum the defects $\delta_n$, while the usual chain-rule for collision entropy accumulates the $\log d_{R_n}$ contributions. This gives~\eqref{eq:decouple-qTPE}.
\end{proof}

\subsection{Frame-potential flow and spectral gap}\label{app:fp-flow}
For a time-window ensemble $\mathcal{U}_{[u,u+\Delta]}$ generated by the EH Hamiltonian with $k$-local coarse-graining, define the second frame potential
\[
\FP_2(\Delta)\ :=\ \mathbb{E}_{U,V\sim\mathcal{U}_{[u,u+\Delta]}}\left|\operatorname{Tr}(U^\dagger V)\right|^4.
\]
Let $\mathcal{L}^{(2)}$ be the generator of the second-moment semigroup on $(\mathbb{1},\mathsf{F})^\perp$ obtained by coarse-graining in $u$. A standard computation gives
\begin{equation}\label{eq:fp-ode}
\frac{\mathrm{d}}{\mathrm{d}\Delta}\big(\FP_2(\Delta)-\FP^{\rm Haar}_2\big)\ =\ -2\,\mathcal{E}_2\!\left(\sigma_\Delta\right),
\end{equation}
where $\mathcal{E}_2$ is the Dirichlet form of the symmetrized generator on $(\mathbb{1},\mathsf{F})^\perp$ and $\sigma_\Delta$ the centered second-moment state. If the Poincar\'e (spectral-gap) inequality
\begin{equation}\label{eq:poincare}
\mathcal{E}_2(X)\ \ge\ \Gamma\,\|X\|_2^2\qquad\text{for some }\Gamma>0,
\end{equation}
holds on the support of $\sigma_\Delta$, then~\eqref{eq:fp-ode} implies exponential relaxation
\[
\FP_2(\Delta)-\FP^{\rm Haar}_2\ \le\ e^{-2\Gamma\Delta}\,\big(\FP_2(0)-\FP^{\rm Haar}_2\big).
\]
Since $\FP_2-\FP^{\rm Haar}_2$ majorizes $\|\Mtwo-\Pi_{\rm Haar}\|_{2\to 2}^2$ on $(\mathbb{1},\mathsf{F})^\perp$, we obtain the qTPE bound $\delta(\Delta)\le e^{-\Gamma\Delta}$ with $\gamma(\Delta)\ge 1-e^{-\Gamma\Delta}$.

\subsection{From OTOC decay to a gap on $(\mathbb{1},\mathsf{F})^\perp$}\label{app:chaos-gap}
Assume uniform OTOC decay for $k$-local probes within a ballistic light cone of speed $v_B$:
\begin{equation}\label{eq:otoc}
\sup_{\substack{\operatorname{supp}O,V\text{ $k$-local}\\ \operatorname{dist}(\operatorname{supp}O,\operatorname{supp}V)\ge v_B\Delta}}\Big|\langle O^\dagger(t) V^\dagger O(t)V\rangle - \langle O^\dagger O\rangle\langle V^\dagger V\rangle\Big|
\ \le\ C_0\,e^{-2\lambda_L\Delta}\,.
\end{equation}
Using the identity between connected OTOCs and matrix elements of $\Mtwo$ on $(\mathbb{1},\mathsf{F})^\perp$ (via the doubled Heisenberg representation),~\eqref{eq:otoc} implies a uniform contraction of the symmetrized second-moment semigroup on the subspace spanned by $k$-local tensors. A standard comparison argument extends the bound to the full $(\mathbb{1},\mathsf{F})^\perp$ with leakage suppressed as $e^{-\rm const\cdot \Delta v_B/R_{\rm loc}}$, yielding~\eqref{eq:gamma-lower} with $\Gamma\simeq 2\lambda_L$ up to constants.

\subsection{Small $I(R_n{:}E_n)$ and recovery}\label{app:En-recovery}
We model the $n$th step by $U_n=\exp(iH_{\rm int}^{(n)})$ with $\|H_{\rm int}^{(n)}\|\le \kappa_n$ and average over a $(\gamma_n,2)$ qTPE $\nu_n$ prior to the coupling. Working to second order in $\kappa_n$ and using the resolvent representation,
\[
\left(\mathbb{1}-\Mtwo_{\nu_n}\right)^{-1}\Big|_{(\mathbb{1},\mathsf{F})^\perp}\ \le\ \frac{1}{\gamma_n}\,\,
\]
we bound the $\chi^2$-divergence between $\rho_{R_nE_n}$ and $\rho_{R_n}\otimes\rho_{E_n}$ by $O(\kappa_n^2/\gamma_n)$. Pinsker then implies $\|\rho_{R_nE_n}-\rho_{R_n}\otimes\rho_{E_n}\|_1\le C' \sqrt{\kappa_n^2/\gamma_n}$ and Alicki--Fannes gives~\eqref{eq:En-mi}. The Fawzi--Renner bound furnishes a recovery channel $\mathcal{R}_{R_n\to R_n E_n}$ with fidelity $1-O(\kappa_n^2/\gamma_n)$.

\subsection{Adiabatic theorem for mixing channels}\label{app:adiabatic-channels}
Let $\mathcal{T}_u$ be a $C^1$ path of primitive CPTP maps with spectral gap $g>0$ and projectors $\Pi_u$ onto their fixed-point lines. Writing the product map as a time-ordered exponential of generators on the orthogonal complement, a Duhamel expansion combined with a resolvent bound $\|(\mathbb{1}-\mathcal{T}_u)^{-1}\|_{2\to 2}\le 1/g$ yields
\[
\left\|\prod_j\mathcal{T}_{u_j}-\prod_j\Pi_{u_j}\right\|_{2\to 2}\ \le\ \frac{C}{g}\,\sup_u\|\partial_u\mathcal{T}_u\|_{2\to 2}\cdot (u_1-u_0)\ +\ O(\Delta u),
\]
and hence~\eqref{eq:adiabatic-channel} by norm domination. This provides the rigorous version of the non-adiabatic error budget used in the main text.

\subsection{Area--memory upper bound without P0}\label{app:area-memory}
We view $d_{\rm mem}(u)$ as the supremum, over encodings on the interior side of $\Sigma(u)$, of the one-shot entanglement-assisted quantum capacity $\mathcal{Q}^{(1)}_{\rm EA}$ of the induced horizon channel to the interior wedge. For any such encoding,
\[
\log d_{\rm mem}(u)\ \le\ \sup_\rho I_{\rm coh}^{(1)}\big(\text{in}\!\rightarrow\!\text{int}\big)_\rho\ =\ \sup_\rho \Big(-S(\text{in}|\text{int})_\rho\Big),
\]
where $I_{\rm coh}^{(1)}$ is the one-shot coherent information. Let $\sigma$ denote the reference state obtained by vacuum-flow along the generators. The generalized second law (GSL) and monotonicity of relative entropy imply a vacuum-subtracted inequality of the form
\[
S(\text{out}|\text{int})_\rho - S(\text{out}|\text{int})_\sigma\ \ge\ -\frac{A(\Sigma(u))}{4G\hbar}\ +\ O\!\big(\log A(\Sigma(u))\big)\!,
\]
where the additive term accounts for one-shot smoothing/AEP. Rearranging and maximizing over $\rho$ gives~\eqref{eq:area-memory-upper}. (A fully detailed argument can be given using modular-energy first laws and one-shot entropic continuity.)

\subsection{Consequences for the Comb Page analysis}\label{app:comb-consequences}
(i) Replace every use of \Ptwo/\PtwoPrime\ with the verified qTPE gap $\gamma(\Delta)$ from Theorem~\ref{thm:chaos-to-gap}, and propagate errors with Theorem~\ref{thm:qtpe-page}. (ii) Use Theorem~\ref{thm:En-mi} to justify the operational neglect of $E_n$ and to construct an explicit recovery map. (iii) Control non-adiabatic bursts via Theorem~\ref{thm:adiabatic-channel}. (iv) Enforce the coding-rate ceiling of Theorem~\ref{thm:area-memory-upper}.
